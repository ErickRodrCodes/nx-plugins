import { afterAll, beforeAll, describe, expect, it } from 'vitest';
import { WorkspaceGenerator } from '../shared/workspace-generator';

describe('Icon Generation', () => {
  let workspaceGenerator: WorkspaceGenerator;

  beforeAll(async () => {
    workspaceGenerator = new WorkspaceGenerator();
    // Setup complete workspace with Electron app
    await workspaceGenerator.createWorkspace();
    workspaceGenerator.generateReactApp();
    workspaceGenerator.installPluginLocally();
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:init');
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:setup-project guest-app');
  });

  afterAll(async () => {
    await workspaceGenerator.cleanup();
  });

  it('should build icons with build-icons executor', () => {
    console.log('Building icons...');
    workspaceGenerator.execCommand('npx nx build-icons guest-app-electron');

    // Verify icons directory exists
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/src/resources/icon')).toBe(true);
  });

  it('should generate icon files in correct format', () => {
    // Check for generated icon files
    const iconDir = 'apps/guest-app-electron/src/resources/icon';
    expect(workspaceGenerator.fileExists(`${iconDir}/16x16.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/32x32.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/48x48.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/64x64.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/128x128.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/256x256.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/512x512.png`)).toBe(true);
  });

  it('should generate platform-specific icons', () => {
    const iconDir = 'apps/guest-app-electron/src/resources/icon';

    // Check for Windows ICO file
    expect(workspaceGenerator.fileExists(`${iconDir}/icon.ico`)).toBe(true);

    // Check for macOS ICNS file
    expect(workspaceGenerator.fileExists(`${iconDir}/icon.icns`)).toBe(true);

    // Check for Linux PNG files
    expect(workspaceGenerator.fileExists(`${iconDir}/16x16.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/32x32.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/48x48.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/64x64.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/128x128.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/256x256.png`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${iconDir}/512x512.png`)).toBe(true);
  });

  it('should have source icon files', () => {
    // Check for source icon files
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/src/resources/icon/source/icon.png')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/src/resources/icon/source/setup.png')).toBe(true);
  });

  it('should handle icon generation with custom source', () => {
    // Test icon generation with custom source path
    workspaceGenerator.execCommand('npx nx build-icons guest-app-electron --source=apps/guest-app-electron/src/resources/icon/source/custom-icon.png');

    // Verify icons were still generated
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/src/resources/icon/16x16.png')).toBe(true);
  });

  it('should have correct icon configuration in electron-builder.yml', () => {
    // Check electron-builder configuration
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/src/electron-builder.yml')).toBe(true);

    const builderConfig = workspaceGenerator.execCommand('cat apps/guest-app-electron/src/electron-builder.yml', { stdio: 'pipe' });

    // Check for icon configuration
    expect(builderConfig).toContain('icon');
  });
});
