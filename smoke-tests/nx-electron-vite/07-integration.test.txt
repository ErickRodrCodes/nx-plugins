import { afterAll, beforeAll, describe, expect, it } from 'vitest';
import { WorkspaceGenerator } from '../shared/workspace-generator';

describe('Complete Integration Workflow', () => {
  let workspaceGenerator: WorkspaceGenerator;

  beforeAll(() => {
    workspaceGenerator = new WorkspaceGenerator();
  });

  afterAll(async () => {
    await workspaceGenerator.cleanup();
  });

  it('should complete the full nx-electron-vite workflow successfully', async () => {
    // Step 1: Create workspace and setup
    console.log('Step 1: Creating Nx workspace...');
    workspaceGenerator.createWorkspace();
    workspaceGenerator.generateReactApp();
    workspaceGenerator.installPluginLocally();

    // Step 2: Initialize plugin
    console.log('Step 2: Initializing plugin...');
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:init');

    // Step 3: Setup Electron project
    console.log('Step 3: Setting up Electron project...');
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:setup-project guest-app');

    // Step 4: Build native modules
    console.log('Step 4: Building native modules...');
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:build-native');

    // Step 5: Generate icons
    console.log('Step 5: Generating icons...');
    workspaceGenerator.execCommand('npx nx build-icons guest-app-electron');

    // Step 6: Build the app
    console.log('Step 6: Building Electron app...');
    workspaceGenerator.execCommand('npx nx build-electron guest-app-electron');

    // Step 7: Verify complete output
    console.log('Step 7: Verifying outputs...');

    // Verify workspace structure
    expect(workspaceGenerator.fileExists('apps/guest-app')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron')).toBe(true);

    // Verify Electron app structure
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/src/main/main.ts')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/src/preload/preload.ts')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/index.html')).toBe(true);

    // Verify configuration files
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/electron-nx-vite.config.ts')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/vite.config.ts')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/tsconfig.app.json')).toBe(true);

    // Verify build output
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist/main.js')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist/preload.js')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist/index.html')).toBe(true);

    // Verify resources
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist/resources')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist/resources/icon')).toBe(true);

    // Verify native modules
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist/native')).toBe(true);

    // Verify package.json in build output
    const packageJson = workspaceGenerator.readJsonFile('apps/guest-app-electron/dist/package.json');
    expect(packageJson.name).toBeDefined();
    expect(packageJson.version).toBeDefined();
    expect(packageJson.main).toBeDefined();

    console.log('✅ Complete integration workflow passed!');
  }, 300000); // 5 minute timeout for full workflow

  it('should handle workspace with different presets', () => {
    // Test with different workspace preset
    workspaceGenerator.cleanup();
    workspaceGenerator.createWorkspace({ preset: 'react' });
    workspaceGenerator.installPluginLocally();
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:init');
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:setup-project my-react-app');

    // Verify it works with React preset
    expect(workspaceGenerator.fileExists('apps/my-react-app-electron')).toBe(true);
    expect(workspaceGenerator.fileExists('apps/my-react-app-electron/project.json')).toBe(true);
  });

  it('should handle different package managers', () => {
    // Test with npm package manager
    workspaceGenerator.cleanup();
    workspaceGenerator.createWorkspace({ packageManager: 'npm' });
    workspaceGenerator.generateReactApp();
    workspaceGenerator.installPluginLocally();
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:init');
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:setup-project guest-app');

    // Verify it works with npm
    expect(workspaceGenerator.fileExists('apps/guest-app-electron')).toBe(true);
  });
});
