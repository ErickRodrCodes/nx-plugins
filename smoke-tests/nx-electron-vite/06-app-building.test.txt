import { afterAll, beforeAll, describe, expect, it } from 'vitest';
import { WorkspaceGenerator } from '../shared/workspace-generator';

describe('Electron App Building', () => {
  let workspaceGenerator: WorkspaceGenerator;

  beforeAll(async () => {
    workspaceGenerator = new WorkspaceGenerator();
    // Setup complete workspace with Electron app
    await workspaceGenerator.createWorkspace();
    workspaceGenerator.generateReactApp();
    workspaceGenerator.installPluginLocally();
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:init');
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:setup-project guest-app');
    workspaceGenerator.execCommand('npx nx g nx-electron-vite:build-native');
    workspaceGenerator.execCommand('npx nx build-icons guest-app-electron');
  });

  afterAll(async () => {
    await workspaceGenerator.cleanup();
  });

  it('should build the Electron app with build-electron executor', () => {
    console.log('Building Electron app...');
    workspaceGenerator.execCommand('npx nx build-electron guest-app-electron');

    // Verify build output exists
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist')).toBe(true);
  });

  it('should generate correct build output structure', () => {
    const distDir = 'apps/guest-app-electron/dist';

    // Check for main build files
    expect(workspaceGenerator.fileExists(`${distDir}/main.js`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${distDir}/preload.js`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${distDir}/index.html`)).toBe(true);
  });

  it('should include native modules in build output', () => {
    const distDir = 'apps/guest-app-electron/dist';

    // Check for native modules directory
    expect(workspaceGenerator.fileExists(`${distDir}/native`)).toBe(true);

    // Check for readme in native directory
    expect(workspaceGenerator.fileExists(`${distDir}/native/readme.md`)).toBe(true);
  });

  it('should include resources in build output', () => {
    const distDir = 'apps/guest-app-electron/dist';

    // Check for resources directory
    expect(workspaceGenerator.fileExists(`${distDir}/resources`)).toBe(true);
    expect(workspaceGenerator.fileExists(`${distDir}/resources/icon`)).toBe(true);
  });

  it('should have correct package.json in build output', () => {
    const distDir = 'apps/guest-app-electron/dist';

    expect(workspaceGenerator.fileExists(`${distDir}/package.json`)).toBe(true);

    const packageJson = workspaceGenerator.readJsonFile(`${distDir}/package.json`);

    // Verify essential fields
    expect(packageJson.name).toBeDefined();
    expect(packageJson.version).toBeDefined();
    expect(packageJson.main).toBeDefined();
    expect(packageJson.scripts).toBeDefined();
  });

  it('should handle development build', () => {
    console.log('Building Electron app in development mode...');
    workspaceGenerator.execCommand('npx nx build-electron guest-app-electron --configuration=development');

    // Verify build output exists
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist')).toBe(true);
  });

  it('should handle production build', () => {
    console.log('Building Electron app in production mode...');
    workspaceGenerator.execCommand('npx nx build-electron guest-app-electron --configuration=production');

    // Verify build output exists
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/dist')).toBe(true);
  });

  it('should have correct Vite build configuration', () => {
    // Check Vite config
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/vite.config.ts')).toBe(true);

    const viteConfig = workspaceGenerator.execCommand('cat apps/guest-app-electron/vite.config.ts', { stdio: 'pipe' });

    // Check for Electron-specific configuration
    expect(viteConfig).toContain('defineConfig');
  });

  it('should have correct electron-nx-vite configuration', () => {
    // Check electron-nx-vite config
    expect(workspaceGenerator.fileExists('apps/guest-app-electron/electron-nx-vite.config.ts')).toBe(true);

    const config = workspaceGenerator.execCommand('cat apps/guest-app-electron/electron-nx-vite.config.ts', { stdio: 'pipe' });

    // Check for configuration structure
    expect(config).toContain('export default');
  });
});
